# cello-note-detector
hackathon app for identifying cello notes using audio and video

PYTHON 3.11.9

cd cello-note-detector
python -m venv venv

Windows:
venv\Scripts\activate

Mac:
source venv/bin/activate

pip install torch torchaudio
pip install transformers librosa soundfile
pip install numpy matplotlib
